cmake_minimum_required(VERSION 3.18)
project(output_format_tests LANGUAGES CXX)
include(CTest)
add_subdirectory(../../src/io trajectory_io)
if(USE_BXYZ)
  add_executable(test_raw test_raw.cpp)
  target_include_directories(test_raw PRIVATE ../../src/third_party/binary_xyz/cpp/include)

  target_link_libraries(test_raw PRIVATE trajectory_io)
  add_test(NAME raw COMMAND test_raw "${CMAKE_CURRENT_BINARY_DIR}/frames")
else()
  add_executable(test_text test_text.cpp)
  target_link_libraries(test_text PRIVATE trajectory_io)
  add_test(NAME text_config COMMAND test_text)
endif()
