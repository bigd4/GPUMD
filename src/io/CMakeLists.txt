option(USE_BXYZ "Enable BXYZ/ZBXYZ input and output" OFF)
add_library(trajectory_io STATIC output_format.cpp)
target_include_directories(trajectory_io PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
set_target_properties(trajectory_io PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
if(USE_BXYZ)
  set(BXYZ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/binary_xyz")
  list(APPEND CMAKE_MODULE_PATH "${BXYZ_ROOT}/cpp/cmake")
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  find_package(ZSTD REQUIRED)
  target_sources(trajectory_io PRIVATE "${BXYZ_ROOT}/cpp/src/binary_xyz.cpp")
  target_include_directories(trajectory_io PRIVATE "${BXYZ_ROOT}/cpp/include")
  target_compile_definitions(trajectory_io PRIVATE USE_BXYZ)
  set_property(TARGET trajectory_io PROPERTY CXX_STANDARD 17)
  target_link_libraries(trajectory_io PUBLIC OpenSSL::Crypto ZSTD::ZSTD)
endif()
